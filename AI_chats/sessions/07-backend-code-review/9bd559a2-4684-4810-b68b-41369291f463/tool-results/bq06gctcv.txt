    33	def exception_handler(exc: Exception, context: dict[str, Any]) -> Response | None:
    34	    """Render an API error with codes; DRF's own handler renders messages only.
    35	
    36	    :param exc: The exception a view raised.
    37	    :param context: The view, request and URL arguments, as DRF passes them.
    38	    :returns: The error response, or ``None`` for an exception that is not an
    39	        API error, which Django then turns into a 500.

    42	        exc = NotFound()

    45	        details = exc.get_full_details()

    47	    return response

     1	"""Recognising a job id in text that came from a URL."""
     2	
     3	# Job.id is a BigAutoField: a positive 64-bit integer.
     4	MAX_JOB_ID = 2**63 - 1
     5	
     6	
     7	def parse_job_id(text: str) -> int | None:
     8	    """Read a job id written as plain decimal digits.
     9	
    10	    Stricter than ``int()``, which also accepts signs, spaces, underscores and
    11	    non-ASCII digits, and nothing above the column's range is let through to a
    12	    query.
    13	
    14	    :param text: A path segment or a cursor position.
    15	    :returns: The id, or ``None`` if the text is not one a job could have.
    16	    """
    17	    if not (text.isascii() and text.isdigit()) or len(text) > len(str(MAX_JOB_ID)):
    18	        return None
    19	    job_id = int(text)
    20	    return job_id if job_id <= MAX_JOB_ID else None

     1	"""Cursor (keyset) pagination for the job list."""

     9	class JobCursorPagination(CursorPagination):
    10	    """Pages of 20 jobs, newest first.
    11	
    12	    Each page is ``WHERE id < <last id seen> ORDER BY id DESC LIMIT 21``: the
    13	    same cost on the first page and the fifty-thousandth, and never a
    14	    ``COUNT(*)``. The page size is fixed; the caller cannot raise it.
    15	    """
    16	
    17	    page_size = 20
    18	    # Unique, as cursor pagination needs, and already indexed (primary key).
    19	    ordering = "-id"
    20	
    21	    def decode_cursor(self, request: Request) -> Cursor | None:
    22	        """Refuse a cursor whose position is not a job id.
    23	
    24	        DRF checks the offset and the direction but passes the position to the
    25	        query as it came, where text such as ``abc`` raises ``ValueError``: a
    26	        500 from a URL anyone can type.
    27	
    28	        :param request: The list request.
    29	        :returns: The decoded cursor, or ``None`` when the request has none.
    30	        :raises NotFound: If the cursor cannot be decoded or its position is
    31	            not a job id.
    32	        """
    33	        cursor = super().decode_cursor(request)
    34	        if cursor is not None and cursor.position is not None and parse_job_id(cursor.position) is None:
    35	            raise NotFound(self.invalid_cursor_message)
    36	        return cursor

    19	class DuplicateJobName(Exception):
    20	    """A job with this name already exists."""

    22	
    23	def create_job(name: str) -> Job:
    24	    """Create a job together with its initial ``PENDING`` status row.
    25	
    26	    :param name: The job's name; surrounding whitespace is removed.
    27	    :returns: The saved job.
    28	    :raises ValueError: If the name is not a string, is blank, contains a NUL
    29	        character or a lone surrogate, or is longer than the column allows.
    30	    :raises DuplicateJobName: If another job already has this name.
    31	    """
    32	    name = _clean_name(name)
    33	    try:
    34	        # No "does this name exist?" query first: two concurrent calls would
    35	        # both see "no". The unique constraint decides, and this atomic block
    36	        # is the savepoint that undoes the failed insert.
    37	        with transaction.atomic():
    38	            job = Job.objects.create(name=name)
    39	            JobStatus.objects.create(job=job, status_type=job.current_status)
    40	    except IntegrityError as exc:
    41	        if _violated_constraint(exc) == JOB_NAME_UNIQUE_CONSTRAINT:
    42	            raise DuplicateJobName(name) from exc
    43	        raise
    44	    return job

    47	def change_status(job_id: int, status: str) -> Job:
    48	    """Append a status row to a job's log and copy the status onto the job.
    49	
    50	    Any state may follow any state, including the current one; every call
    51	    appends a row.
    52	
    53	    :param job_id: The id of the job to change.
    54	    :param status: One of the ``StatusType`` values.
    55	    :returns: The job, carrying the new ``current_status`` and ``updated_at``.
    56	    :raises ValueError: If the status is not one of the ``StatusType`` values.
    57	    :raises Job.DoesNotExist: If no job has this id.
    58	    """
    59	    new_status = _clean_status(status)
    60	    with transaction.atomic():
    61	        job = _lock_job(job_id)
    62	        JobStatus.objects.create(job=job, status_type=new_status)
    63	        job.current_status = new_status
    64	        # updated_at must be listed: with update_fields, an auto_now field

    73	@contextmanager
    74	def service_errors() -> Iterator[None]:
    75	    """Turn the errors a service raises into API errors.
    76	
    77	    Wrapped around the service call only. A handler for every ``ValueError``
    78	    raised anywhere in a request would report bugs as the client's mistake.
    79	
    80	    :raises ValidationError: 400, for a ``ValueError``.
    81	    :raises NotFound: 404, for ``Job.DoesNotExist``.
    82	    :raises Conflict: 409, for ``DuplicateJobName``.
    83	    """
    84	    try:
    85	        yield
    86	    except ValueError as exc:
    87	        # Reached only if a serializer and its service disagree on a rule.
    88	        raise ValidationError({api_settings.NON_FIELD_ERRORS_KEY: [str(exc)]}) from exc
    89	    except Job.DoesNotExist as exc:
    90	        raise NotFound() from exc
    91	    except DuplicateJobName as exc:
